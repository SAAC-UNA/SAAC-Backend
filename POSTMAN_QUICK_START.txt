Headers:
  Authorization: Bearer {{token}}
  Accept: application/json


┌─────────────────────────────────────────────────────────────┐
│ 4. CREAR SOLICITUD                                          │
└─────────────────────────────────────────────────────────────┘

POST {{base_url}}/solicitudes-ampliacion-tiempo

Headers:
  Authorization: Bearer {{token}}
  Content-Type: application/json
  Accept: application/json

Body (raw JSON):
{
    "evidencia_asignacion_id": 1,
    "motivo": "Solicito ampliación debido a responsabilidades académicas",
    "fecha_sugerida": "2026-06-30"
}

⚠️ VALIDACIONES:
  ✓ evidencia_asignacion_id debe existir
  ✓ motivo mínimo 10 caracteres
  ✓ fecha_sugerida debe ser FUTURA
  ✓ NO puede haber otra solicitud PENDIENTE para misma evidencia


┌─────────────────────────────────────────────────────────────┐
│ 5. ACTUALIZAR SOLICITUD (solo PENDIENTE)                   │
└─────────────────────────────────────────────────────────────┘

PUT {{base_url}}/solicitudes-ampliacion-tiempo/11

Headers:
  Authorization: Bearer {{token}}
  Content-Type: application/json

Body (raw JSON):
{
    "motivo": "Motivo actualizado con más detalles",
    "fecha_sugerida": "2026-07-31"
}

⚠️ Solo funciona si estado = "Pendiente"


┌─────────────────────────────────────────────────────────────┐
│ 6. ELIMINAR SOLICITUD (solo PENDIENTE)                     │
└─────────────────────────────────────────────────────────────┘

DELETE {{base_url}}/solicitudes-ampliacion-tiempo/11

Headers:
  Authorization: Bearer {{token}}
  Accept: application/json


┌─────────────────────────────────────────────────────────────┐
│ 7. EVIDENCIAS PRÓXIMAS A VENCER                            │
└─────────────────────────────────────────────────────────────┘

GET {{base_url}}/solicitudes-ampliacion-tiempo/evidencias/proximas-vencer

Headers:
  Authorization: Bearer {{token}}
  Accept: application/json

═══════════════════════════════════════════════════════════════
  CÓDIGOS DE RESPUESTA
═══════════════════════════════════════════════════════════════

200 OK              → Operación exitosa (GET, PUT, DELETE)
201 Created         → Solicitud creada (POST)
401 Unauthorized    → Token inválido/expirado
404 Not Found       → Solicitud no existe
422 Unprocessable   → Validación falló
500 Server Error    → Error interno (revisar logs)
